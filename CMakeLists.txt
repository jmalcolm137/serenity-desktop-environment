# The Serenity Desktop Environment: one build for every component.
#
#   scripts/build-full-sde.sh            # or, equivalently:
#   cmake -S . -B build -GNinja && cmake --build build
#
# This is the single entry point. The per-component CMakeLists.txt still build on
# their own (a component that needs only one piece does not configure the rest),
# but here they are all configured together: one dependency graph, one
# compile_commands.json, and `cmake --build build --target X` reaches any target.
# Each component's outputs land in build/<Component>/, where the scripts and the
# documentation already look for them.
#
# The framework the SDE builds on (libgui-wayland's Lagom: AK, LibCore, LibGUI,
# LibGfx, LibVT, the windowserver shim) is not a subdirectory: it has its own
# entry point and its own build directory, and the SDE reads and drives it (see
# cmake/SDELagom.cmake). Its `lagom` step builds the framework and the programs
# that live in that tree (the Taskbar, Terminal, FileManager, the app suite).
#
# Built (default): the compositor, the shell, the services, the applets, the
# portal backends, the tools/test clients and the test programs. The applications
# live in Lagom: pass -DSDE_BUILD_APPS=ON to build them too.

cmake_minimum_required(VERSION 3.20)
project(SerenityDesktop C CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# One compile_commands.json for the whole desktop, including the Lagom headers.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if (NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# R6 sanitizers: build the SDE's own targets with ASan/UBSan in a separate tree
# (scripts/build-sanitized.sh). The Lagom framework stays uninstrumented; it is a
# separate build, and the sanitizer still intercepts its allocations and reports
# errors in SDE code. Empty (the default) is the normal build.
set(SDE_SANITIZE "" CACHE STRING "Sanitizer list for SDE targets, e.g. address or address,undefined")
if (SDE_SANITIZE)
    add_compile_options(-fsanitize=${SDE_SANITIZE} -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=${SDE_SANITIZE})
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
include(SDELagom)

# wlroots: the compositor's plumbing (0.20, the release Arch ships; see
# docs/21-r5-protocol-parity-plan.md). The protocol XMLs are generated by its
# build, so a source checkout is read from third_party/wlroots.
if (NOT DEFINED SDE_WLROOTS_SRC)
    if (DEFINED ENV{WLROOTS_SRC})
        set(SDE_WLROOTS_SRC "$ENV{WLROOTS_SRC}")
    else()
        set(SDE_WLROOTS_SRC "${CMAKE_SOURCE_DIR}/third_party/wlroots")
    endif()
endif()
set(SDE_WLROOTS_SRC "${SDE_WLROOTS_SRC}" CACHE PATH "The wlroots checkout (its protocol/ XMLs and pkg-config)" FORCE)
if (EXISTS "${SDE_WLROOTS_SRC}/install/lib/pkgconfig")
    set(ENV{PKG_CONFIG_PATH} "${SDE_WLROOTS_SRC}/install/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
    list(APPEND CMAKE_PREFIX_PATH "${SDE_WLROOTS_SRC}/install")
endif()
set(WLROOTS_PROTOCOL_INCLUDE_DIR "${SDE_WLROOTS_SRC}/build/protocol")

# SDE_ROOT: the components that reference the repository (the applets' icons, the
# Taskbar's compat headers) read it.
set(SDE_ROOT "${CMAKE_SOURCE_DIR}")

# --- The components ---------------------------------------------------------
# Grouped as the documentation groups them (docs/03-engineering-conventions.md).
# Each directory owns its targets and its output subdirectory.

# The compositor.
add_subdirectory(WindowServer)

# The shell's services and dialogs.
add_subdirectory(NotificationServer)
add_subdirectory(AudioServer)
add_subdirectory(RequestServer)
add_subdirectory(WebContent)
add_subdirectory(SQLServer)
add_subdirectory(ImageDecoder)
add_subdirectory(LaunchServer)
add_subdirectory(WifiApplet)
add_subdirectory(BatteryApplet)
add_subdirectory(Applets)

# The portal's LibGUI dialogs (docs/15-xdg-desktop-portal-serenity.md).
add_subdirectory(AccessDialog)
add_subdirectory(AppChooserDialog)
add_subdirectory(FileChooserDialog)
add_subdirectory(PrintDialog)
add_subdirectory(ColorPicker)
add_subdirectory(ScreenCastChooser)
add_subdirectory(ScreenshotOverlay)

# The portable tools, the portal backends and the test clients.
add_subdirectory(tools)

# The test programs.
add_subdirectory(test)

# Building the applications: the same list scripts/build-apps.sh builds, in the
# Lagom tree where they live (they are SerenityOS apps, not SDE components).
if (SDE_BUILD_APPS)
    add_custom_target(apps
        COMMAND "${CMAKE_SOURCE_DIR}/scripts/build-apps.sh"
        COMMENT "Building the SerenityOS application suite (libgui-wayland)"
        USES_TERMINAL
        VERBATIM)
endif()

# A convenience target: everything the session needs to start, next to `all`.
add_custom_target(sde ALL
    COMMENT "The Serenity Desktop Environment is built in ${CMAKE_BINARY_DIR}")
add_dependencies(sde lagom)
